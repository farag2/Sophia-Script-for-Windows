; Config for creating a WinRAR SFX archive of Sophia Script for Windows 11 for WinGet
; https://github.com/farag2/Sophia-Script-for-Windows
; https://github.com/farag2/Sophia-Script-for-Windows/blob/main/Scripts/Building/WinGet_SFX.ps1

; Unpack to a temp folder and delete it after the Setup command completes
TempMode

; Copy folder recursively to Downloads folder
Setup=C:\Windows\System32\WindowsPowerShell\v1.0\powershell.exe -NoProfile -WindowStyle Hidden -Command & {$Downloads = Get-ItemPropertyValue -Path 'HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\User Shell Folders' -Name '{374DE290-123F-4565-9164-39C4925E467B}'; New-Item -Path (Join-Path -Path $Downloads -ChildPath "Sophia_Script_for_Windows_11_vSophiaScriptVersion") -ItemType Directory -Force; Copy-Item -Path (Join-Path -Path $PWD.Path -ChildPath '*') -Destination (Join-Path -Path $Downloads -ChildPath "Sophia_Script_for_Windows_11_vSophiaScriptVersion") -Recurse -Force; Invoke-Item -Path (Join-Path -Path $Downloads -ChildPath "Sophia_Script_for_Windows_11_vSophiaScriptVersion")}

; No GUI while expaning SFX archive
Silent=1
